楽天EC 大福帳 テーブル設計

楽天市場の3NF購入データを加工し、分析用の大福帳を作る。ポイントは1つ —— 大福帳はスタースキーマをJOINして平たくしたものそのもの。star を中間層に組めば、最後に結合すれば大福帳になる。

1全体方針

第3正規化された元データ（SHARED_DB.RAKUTEN_EC_RAW）を加工し、 テーブル定義.xlsx に準拠した大福帳を作る。作り方は2層構成。まず スタースキーマ（ファクト1本＋ディメンション4本）を中間層に組み、それを全部JOINして 大福帳（最終マート）に平坦化する。

2スタースキーマ

中心が fct_purchase_item、周囲を4つのディメンションが囲む。 ★＝この課題で新しく加工して付ける列、◆＝数値の事実（measure）。

設計の肝：割引額は「同一商品（同一 item_url）の最大単価との差」。同じ商品でも取引ごとに単価が違うことが前提なので、単価を商品マスタに入れると破綻する。だから unit_price・discount_amount・discount_flag は取引レベル＝ファクトに置く。

SHARED_DB.RAKUTEN_EC_RAW（読み取り専用・dbt では source rakuten_ec として参照）。

テーブル主キー主な列役割

staging（素通し整形の例）

-- models/rakuten_ec/staging/stg_rakuten__purchases.sql
with source as (
    select * from {{ source('rakuten_ec', 'purchases') }}
)
select
    purchase_id,
    message_id,
    purchased_at::timestamp_tz as purchased_at,  -- +0900 を保持して型付け
    user_id,
    destination_postal_code
from source
-- 購入日/購入月はここで作らない → dim_date。他6表も同様に列名統一のみ

int_categories_flatten（カテゴリ階層の平坦化・再帰CTE）

-- models/rakuten_ec/intermediate/int_categories_flatten.sql
with recursive cat as (
    select * from {{ ref('stg_rakuten__categories') }}
),
climb as (
    -- anchor: 起点そのもの
    select category_id as leaf_id,
           category_id, category_name, category_level, parent_category_id
    from cat
    union all
    -- 親カテゴリへ1段ずつ遡る
    select c.leaf_id,
           p.category_id, p.category_name, p.category_level, p.parent_category_id
    from climb c
    join cat p on p.category_id = c.parent_category_id
)
select
    leaf_id as category_id,
    max(case when category_level = 1 then category_name end) as category_level_1,
    max(case when category_level = 2 then category_name end) as category_level_2,
    max(case when category_level = 3 then category_name end) as category_level_3,
    max(case when category_level = 4 then category_name end) as category_level_4
from climb
group by leaf_id

dim_user（年代＝範囲シード / 地方＝都道府県シード）

-- models/rakuten_ec/marts/dim_user.sql
select
    {{ dbt_utils.generate_surrogate_key(['u.user_id']) }} as user_key,
    u.user_id           as user_id_hash,
    u.gender_name,
    u.age,
    g.gene              as age_category,   -- 年代（範囲シード gene_definition）
    u.state_name,                          -- 都道府県（出力名どおり）
    r.region_name,                         -- 地方（state_name に倣う）
    u.marriage_status,
    u.profession_name,
    u.occupation_name
from {{ ref('stg_rakuten__users') }} u
left join {{ ref('gene_definition') }} g
       on u.age between g.age_lower_limit and g.age_upper_limit
left join {{ ref('prefecture_region') }} r      -- seed: state_name, region_name
       on r.state_name = u.state_name

dim_item（名寄せ＋カテゴリ結合＋NULL除外）

-- models/rakuten_ec/marts/dim_item.sql
with items as (
    select * from {{ ref('stg_rakuten__items') }}
),
cleaned as (
    select
        item_id, item_url, category_id, item_name,
        trim(regexp_replace(replace(
            regexp_replace(item_name,
                '^(＼[^／]*／|[0-9]+円以上からご注文可能)[[:space:]]*', ''),  -- セール枠等を除去
            '　', ' '),                          -- 全角スペース→半角
            '[[:space:]]+', ' ')) as item_name_clean  -- 連続空白→1個
    from items
),
name_freq as (
    select item_url, item_name_clean, count(*) as cnt
    from cleaned group by 1, 2
),
ranked as (
    select c.*, f.cnt,
        row_number() over (
            partition by c.item_url
            order by f.cnt desc, length(c.item_name_clean) desc, c.item_id
        ) as rn
    from cleaned c
    join name_freq f
      on f.item_url = c.item_url and f.item_name_clean = c.item_name_clean
)
select
    {{ dbt_utils.generate_surrogate_key(['r.item_url']) }} as item_key,
    r.item_url,
    r.item_name_clean as item_name,          -- 名寄せ後の代表名
    cat.category_level_1, cat.category_level_2,
    cat.category_level_3, cat.category_level_4
from ranked r
left join {{ ref('int_categories_flatten') }} cat
       on cat.category_id = r.category_id
where r.rn = 1
  and cat.category_level_1 is not null       -- カテゴリ1 NULL の商品を除外

fct_purchase_item（measure＋各キー、割引フラグは生の名前から）

-- models/rakuten_ec/marts/fct_purchase_item.sql
with pi as (select * from {{ ref('stg_rakuten__purchase_items') }}),
     p  as (select * from {{ ref('stg_rakuten__purchases') }}),
     it as (select * from {{ ref('stg_rakuten__items') }})   -- 生の item_name（割引フラグ用）
select
    pi.purchase_item_id,
    pi.purchase_id,
    p.message_id,
    p.purchased_at,
    to_number(to_char(p.purchased_at::date, 'YYYYMMDD'))     as date_key,
    {{ dbt_utils.generate_surrogate_key(['p.user_id']) }}    as user_key,
    {{ dbt_utils.generate_surrogate_key(['it.item_url']) }}  as item_key,
    {{ dbt_utils.generate_surrogate_key(['it.store_id']) }}  as store_key,
    p.destination_postal_code,
    pi.unit_price,
    pi.amount,
    pi.unit_price * pi.amount                                as total_price,
    it.item_name ilike '%セール%'                            as discount_flag,   -- 生の名前で判定
    max(pi.unit_price) over (partition by it.item_url) - pi.unit_price as discount_amount
from pi
join p  on p.purchase_id = pi.purchase_id
join it on it.item_id    = pi.item_id

mart_rakuten_ec（大福帳・28列に整形）

-- models/rakuten_ec/marts/mart_rakuten_ec.sql
select
    f.purchase_item_id  as id,
    f.message_id,
    f.purchased_at,
    s.ec_site_name,
    f.unit_price,
    f.amount,
    f.total_price,
    u.user_id_hash,
    i.item_name,
    i.item_url,
    f.destination_postal_code,
    s.store_name,
    u.gender_name,
    u.age,
    u.age_category,
    u.state_name,
    u.marriage_status,
    u.profession_name,
    u.occupation_name,
    i.category_level_1, i.category_level_2, i.category_level_3, i.category_level_4,
    d.purchase_date,
    d.purchase_month,
    u.region_name,
    f.discount_flag,
    f.discount_amount
from {{ ref('fct_purchase_item') }} f
join {{ ref('dim_item') }}  i on i.item_key  = f.item_key    -- inner join → カテゴリ1 NULL商品は除外
join {{ ref('dim_user') }}  u on u.user_key  = f.user_key
join {{ ref('dim_store') }} s on s.store_key = f.store_key
join {{ ref('dim_date') }}  d on d.date_key  = f.date_key
